cmake_minimum_required(VERSION 3.16)
project(tess VERSION 0.1.0 LANGUAGES C)

function(tess_c11 target)
  set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()

add_library(tess
  src/tess.c src/exact.c src/geometry.c src/api.c src/predicates.c src/audit.c
  src/legalize.c src/topology.c src/snapshot.c src/overlay.c src/certificate.c
  src/metrics.c src/edits.c src/remesh.c src/bounds.c src/features.c)
target_include_directories(tess PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
tess_c11(tess)
if(UNIX)
  target_link_libraries(tess PUBLIC m)
endif()
install(TARGETS tess ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(FILES include/tess.h DESTINATION include)

set(TESS_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(TESS_OBJ_DEFAULT OFF)
if(EXISTS "${TESS_REPO}/tiny_obj_c.c")
  set(TESS_OBJ_DEFAULT ON)
endif()
option(TESS_BUILD_OBJ "Build repository tinyobjloader OBJ adapter" ${TESS_OBJ_DEFAULT})
if(TESS_BUILD_OBJ)
  if(NOT EXISTS "${TESS_REPO}/tiny_obj_c.c" OR NOT EXISTS "${TESS_REPO}/tobj_write.c")
    message(FATAL_ERROR "TESS_BUILD_OBJ needs repository tinyobjloader C11 loader/writer sources")
  endif()
  add_executable(tess_obj tools/obj.c tools/obj_support.c
    tools/obj_loader.c tools/obj_number.c ${TESS_REPO}/tobj_tess.c ${TESS_REPO}/tobj_write.c)
  add_executable(tess_obj_audit tools/obj_audit.c tools/obj_loader.c tools/obj_number.c ${TESS_REPO}/tobj_tess.c)
  target_include_directories(tess_obj_audit PRIVATE src ${TESS_REPO})
  target_compile_definitions(tess_obj_audit PRIVATE TOBJ_ENABLE_FILE_IO)
  target_link_libraries(tess_obj_audit PRIVATE tess)
  set_target_properties(tess_obj_audit PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  target_include_directories(tess_obj PRIVATE src ${TESS_REPO})
  target_compile_definitions(tess_obj PRIVATE TOBJ_ENABLE_FILE_IO)
  target_link_libraries(tess_obj PRIVATE tess)
  set_target_properties(tess_obj PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    set_source_files_properties(tools/obj.c tools/obj_support.c tools/obj_audit.c tools/obj_number.c tools/obj_loader.c
      PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Wpedantic;-Werror")
  endif()
endif()

include(CTest)
option(TESS_TEST_LEAN "Check actual C artifacts with the built Lean checker" OFF)
set(TESS_LEAN_CHECKER "${TESS_REPO}/formal/.lake/build/bin/tesscheck"
  CACHE FILEPATH "Executable Lean certificate checker")
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter)
  function(tess_test target test source)
    add_executable(${target} ${source})
    target_include_directories(${target} PRIVATE src)
    target_link_libraries(${target} PRIVATE tess)
    tess_c11(${target})
    if(NOT "${test}" STREQUAL "")
      add_test(NAME ${test} COMMAND ${target})
    endif()
  endfunction()
  tess_test(tess_test tess_predicate tests/test_predicate.c)
  if(TESS_BUILD_OBJ)
    tess_test(tess_obj_number_test tess_obj_number tests/test_obj_number.c)
    target_sources(tess_obj_number_test PRIVATE tools/obj_number.c)
    target_include_directories(tess_obj_number_test PRIVATE tools)
  endif()
  foreach(name geometry audit resources overlay certificate metrics remesh broadphase rollback features diagnostics rounding fuzz adversarial)
    tess_test(tess_${name}_test tess_${name} tests/test_${name}.c)
  endforeach()
  tess_test(tess_arithmetic_core_probe "" tests/arithmetic_core_probe.c)
  tess_test(tess_exact_probe "" tests/exact_probe.c)
  tess_test(tess_predicate_probe "" tests/predicate_probe.c)
  tess_test(tess_emit_certificate "" tests/emit_certificate.c)
  tess_test(tess_benchmark "" tools/benchmark.c)
  tess_test(tess_resource_sweep "" tools/resource_sweep.c)
  if(Python3_FOUND)
    add_test(NAME tess_segment_oracle COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/segment_oracle.py $<TARGET_FILE:tess_predicate_probe>)
    add_test(NAME tess_arithmetic_core_oracle COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/arithmetic_core_oracle.py $<TARGET_FILE:tess_arithmetic_core_probe>)
    add_test(NAME tess_benchmark_gate COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/benchmark_regression.py)
    add_test(NAME tess_exact_oracle COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/oracle.py $<TARGET_FILE:tess_exact_probe>)
    add_test(NAME tess_predicate_oracle COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/predicate_oracle.py $<TARGET_FILE:tess_predicate_probe>)
    if(TESS_BUILD_OBJ)
      add_test(NAME tess_obj_regression COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/obj_regression.py $<TARGET_FILE:tess_obj>)
      set(TESS_FUZZ_CHECKER_ARGS "")
      if(TESS_TEST_LEAN)
        list(APPEND TESS_FUZZ_CHECKER_ARGS --checker ${TESS_LEAN_CHECKER})
      endif()
      add_test(NAME tess_parser_fuzz COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/fuzz_regression.py $<TARGET_FILE:tess_obj>
        $<TARGET_FILE:tess_obj_audit> $<TARGET_FILE:tess_emit_certificate> ${TESS_FUZZ_CHECKER_ARGS})
    endif()
  endif()
  if(TESS_TEST_LEAN)
    if(NOT Python3_FOUND OR NOT EXISTS "${TESS_LEAN_CHECKER}")
      message(FATAL_ERROR "TESS_TEST_LEAN requires Python and a built tesscheck executable")
    endif()
    add_test(NAME tess_lean_certificates COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/lean_regression.py $<TARGET_FILE:tess_emit_certificate> ${TESS_LEAN_CHECKER})
    if(TESS_BUILD_OBJ)
      add_test(NAME tess_certified_assets COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/asset_regression.py $<TARGET_FILE:tess_obj> ${TESS_LEAN_CHECKER} $<TARGET_FILE:tess_obj_audit>)
    endif()
  endif()
endif()
